use ebook_rs::Book;

fn main() -> Result<(), String> {
    // Auto-detects EPUB, MOBI, AZW3, FB2, KEPUB, or LIT from path
    let mut book = Book::from_file("alice.azw3")?;

    println!("Title: {}", book.metadata().title);
    println!("Total Sections: {}", book.spine().len());

    // Search full-text across chapters
    let matches = book.search("Rabbit");
    println!("Found {} search matches", matches.len());

    Ok(())
}
